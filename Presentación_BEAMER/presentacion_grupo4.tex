\section{2. Problemática}

% 3
\begin{frame}{La moderación de YouTube combina escala y contexto}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.32\textwidth}
      \begin{beamercolorbox}[wd=\linewidth,sep=3mm,rounded=true]{epoch2}
        \pill{Coral}{Contenido dañino}\par\smallskip
        \small Violencia, acoso, discriminación, ataques por identidad y sexualización pueden aparecer en redes y videos.
      \end{beamercolorbox}
    \end{column}
    \begin{column}{.32\textwidth}
      \begin{beamercolorbox}[wd=\linewidth,sep=3mm,rounded=true]{epoch3}
        \pill{Gold}{Contexto}\par\smallskip
        \small Una noticia puede citar una amenaza; el humor puede encubrir un blanco; una expresión local cambia el sentido.
      \end{beamercolorbox}
    \end{column}
    \begin{column}{.32\textwidth}
      \begin{beamercolorbox}[wd=\linewidth,sep=3mm,rounded=true]{epoch2}
        \pill{Blue}{Decisión}\par\smallskip
        \small Revisar el video completo no escala. Una alerta sin texto ni tiempo tampoco permite auditar la sugerencia.
      \end{beamercolorbox}
    \end{column}
  \end{columns}
  \vspace{4mm}
  \takeaway{El problema no es solo clasificar texto: es priorizar sin perder contexto, incertidumbre, procedencia ni responsabilidad humana.}
  \sourceNote{Gorwa, Binns y Katzenbach (2020); Waseem, Davidson, Warmsley y Weber (2017); Banko et al. (2020); Defensoría del Pueblo (2021).}
\end{frame}

\begin{frame}{El problema se manifiesta en tres niveles}

  \resizebox{.98\textwidth}{!}{\input{figura_situacion_beamer.tex}}

\begin{frame}{Las brechas específicas delimitan la construcción}
  \centering
  \begin{tikzpicture}[node distance=2.4mm,
    b/.style={draw,rounded corners=3pt,text width=.88\textwidth,minimum height=10mm,align=left,inner sep=2.3mm,font=\scriptsize}]
    \node[b,draw=Blue!45,fill=Blue!7] (b1) {\textbf{Datos dispersos.} Los subtítulos públicos carecen de una adquisición, depuración y segmentación común que preserve video, texto, tiempo y procedencia.};
    \node[b,draw=Gold!50,fill=Gold!9,below=of b1] (b2) {\textbf{Criterios heterogéneos.} Falta aplicar un contrato operacional contextualizado al Perú que distinga daño, seguridad explícita e incertidumbre sin convertir la selección del canal en etiqueta.};
    \node[b,draw=Teal!45,fill=Teal!7,below=of b2] (b3) {\textbf{Etiquetas incompletas.} El historial recuperable y los casos nuevos requieren consolidación incremental, trazable, económica y supervisable.};
    \node[b,draw=Purple!45,fill=Purple!7,below=of b3] (b4) {\textbf{Evidencia no comparable.} Las familias clásicas y neuronales requieren un mismo snapshot, partición por video, calibración y evaluación pareada.};
    \node[b,draw=Coral!45,fill=Coral!7,below=of b4] (b5) {\textbf{Decisión desconectada.} Los scores deben convertirse en evidencia temporal y motivos de revisión, sin automatizar sanciones ni ocultar la intervención humana.};
  \end{tikzpicture}
  \vspace{1mm}
  \takeaway{Son brechas necesarias para construir el artefacto; publicarlo o presentarlo es una tarea de comunicación, no un problema de investigación.}
\end{frame}

\begin{frame}{Límites visibles y trabajo futuro accionable}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.49\textwidth}
      \pill{Coral}{Límites}\par\smallskip
      \begin{itemize}\small
        \item pseudoetiquetas y sugerencia visible: riesgo de anclaje;
        \item sin \emph{gold} humano ciego, doble anotación ni kappa;
        \item muestreo dirigido y subtítulos disponibles, no prevalencia natural;
        \item solo texto: faltan señales visuales y acústicas;
        \item resultados neuronales actuales con una semilla y test históricamente consultado.
      \end{itemize}
    \end{column}
    \begin{column}{.49\textwidth}
      \pill{Teal}{Siguientes iteraciones}\par\smallskip
      \begin{itemize}\small
        \item cerrar Flash, Pro, revisión humana y snapshot v2.1;
        \item ejecutar seis ramas, cinco umbrales y comparación pareada por video;
        \item crear holdout temporal de prevalencia natural con doble revisión;
        \item repetir semillas y pruebas conductuales/adversariales;
        \item medir deriva, subgrupos, latencia, privacidad, seguridad y capacidad humana en modo sombra.
      \end{itemize}
    \end{column}
