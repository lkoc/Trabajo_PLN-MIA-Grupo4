    \begin{column}{.32\textwidth}
      \begin{beamercolorbox}[wd=\linewidth,sep=3mm,rounded=true]{epoch3}

    \begin{column}{.32\textwidth}
      \begin{beamercolorbox}[wd=\linewidth,sep=3mm,rounded=true]{epoch2}

\begin{frame}{Racismo/discriminación exige leer racialización, no palabras aisladas}

\begin{frame}{Acoso, amenaza y sexualidad conservan fronteras distintas}
  \scriptsize
  \renewcommand{\arraystretch}{1.18}
  \begin{tabularx}{\textwidth}{>{\bfseries\raggedright\arraybackslash}p{32mm}X>{\raggedright\arraybackslash}p{50mm}}

\end{frame}
